Schema, JOIN, ushtrime qe i japin ne tabele

Tabela shembull

users(id, name, email, created_at)
orders(id, user_id, total, created_at)
order_items(id, order_id, product_id, qty, price)
products(id, name, stock)

1:N users→orders. N:M orders↔products permes order_items.

PK = id. FK = user_id referon users(id). UNIQUE ne email.

CRUD

INSERT INTO users (name, email) VALUES ('Ana', 'ana@x.com');
SELECT id, email FROM users WHERE email = 'ana@x.com';
UPDATE users SET name = 'Ana K' WHERE id = 1;
DELETE FROM users WHERE id = 1;  -- kujdes FK

JOIN

SELECT u.name, o.total
FROM users u
INNER JOIN orders o ON o.user_id = u.id;

INNER: vetem me porosi. LEFT: te gjithe users.

SELECT u.id, u.name
FROM users u
LEFT JOIN orders o ON o.user_id = u.id
WHERE o.id IS NULL;

Userat pa porosi — e duan shpesh.

GROUP BY

SELECT user_id, COUNT(*) AS n, SUM(total) AS spent
FROM orders
GROUP BY user_id
HAVING SUM(total) > 100
ORDER BY spent DESC;

WHERE filtron rreshta para grupimit. HAVING pas.

Ushtrime — shkruaji pa parë

  1. Emrat e users te krijuar sot.
  2. Totali i porosive per muaj.
  3. Produktet qe nuk jane shitur kurre.
  4. Useri me spent me te larte.
SELECT name FROM users WHERE created_at::date = CURRENT_DATE;
-- MySQL: DATE(created_at) = CURDATE()

SELECT DATE_TRUNC('month', created_at) AS m, SUM(total)
FROM orders GROUP BY 1 ORDER BY 1;

SELECT p.name
FROM products p
LEFT JOIN order_items i ON i.product_id = p.id
WHERE i.id IS NULL;

SELECT u.email, SUM(o.total) AS spent
FROM users u
JOIN orders o ON o.user_id = u.id
GROUP BY u.email
ORDER BY spent DESC
LIMIT 1;
