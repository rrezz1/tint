W3. FastAPI PATCH done

from typing import Optional
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()
TASKS = {1: {"id": 1, "title": "x", "done": False}}

class TaskPatch(BaseModel):
    title: Optional[str] = None
    done: Optional[bool] = None

@app.patch("/tasks/{tid}")
def patch_task(tid: int, body: TaskPatch):
    if tid not in TASKS:
        raise HTTPException(404, "not found")
    data = body.model_dump(exclude_unset=True)
    TASKS[tid].update(data)
    return TASKS[tid]

W5. Pagination me offset

page fillon nga 1, size 20. Shkruaj slice Python.

def page_slice(items, page, size=20):
    start = (page - 1) * size
    return {
        "count": len(items),
        "results": items[start:start + size],
    }

async function loadUsers() {
  const res = await fetch("/api/users");
  if (!res.ok) throw new Error("fail " + res.status);
  const users = await res.json();

W7. JS: POST form pa reload

form.addEventListener("submit", async (e) => {
  e.preventDefault();
  const body = {
    email: form.email.value,
    name: form.name.value,
  };
  const res = await fetch("/api/users", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  if (res.status === 201) location.reload();
});
