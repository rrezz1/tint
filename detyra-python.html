        raise ValueError("n negativ")
    out = 1

P4. Fibonacci n-te

0, 1, 1, 2, 3, 5… kthe F(n).

def fib(n):
    if n < 0:
        raise ValueError("n")
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

Mos bej rekursion te dyfishte pa memo — O(2^n).

P7. Parentheses valide

"()[]{}" True. "(]" False.

def valid(s):
    pair = {")": "(", "]": "[", "}": "{"}
    st = []
    for ch in s:
        if ch in "([{":
            st.append(ch)
        elif ch in pair:
            if not st or st[-1] != pair[ch]:
                return False
            st.pop()
    return not st

def top_word(text):

P13. Merge dy lista te renditura

def merge(a, b):
    i = j = 0
    out = []
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            out.append(a[i]); i += 1
        else:
            out.append(b[j]); j += 1
    return out + a[i:] + b[j:]

Mos sorted(a+b) nese duan O(n) me pointer.

P15. Klasa BankAccount

deposit, withdraw (mos lejo minus), balance property.

class BankAccount:
    def __init__(self):
        self._balance = 0

    @property
    def balance(self):
        return self._balance

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("amount")
        self._balance += amount

    def withdraw(self, amount):
        if amount <= 0 or amount > self._balance:
            raise ValueError("funds")
        self._balance -= amount

P17. Dedup email (case insensitive)

def unique_emails(emails):
    seen = set()
    out = []
    for e in emails:
        key = e.lower()
        if key not in seen:
            seen.add(key)
            out.append(e)
    return out

P18. Retry 3 here

Therret fn(). Nese hedh, provo perseri. Nese 3 fail, raise.

def retry(fn, times=3):
    last = None
    for _ in range(times):
        try:
            return fn()
        except Exception as e:
            last = e
    raise last

Urdher per API te brishte. Fol: mos fshih gabimin perherë.
