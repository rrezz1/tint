16 detyra SQL + zgjidhje

Schema:

users(id, name, email, created_at)
orders(id, user_id, total, status, created_at)
order_items(id, order_id, product_id, qty, price)
products(id, name, stock)

S1. Te gjithe users, vetem id dhe email

SELECT id, email FROM users;

S2. Users me email @gmail.com

SELECT * FROM users WHERE email ILIKE '%@gmail.com';
-- MySQL: LOWER(email) LIKE '%@gmail.com'

S3. Porosi mbi 50, te renditura zbritje

SELECT * FROM orders WHERE total > 50 ORDER BY total DESC;

S4. Emri i user + totali i porosisë

SELECT u.name, o.total
FROM users u
INNER JOIN orders o ON o.user_id = u.id;

S5. Users pa asnje porosi

SELECT u.*
FROM users u
LEFT JOIN orders o ON o.user_id = u.id
WHERE o.id IS NULL;

S6. Sa porosi ka secili user (edhe 0)

SELECT u.id, u.name, COUNT(o.id) AS n
FROM users u
LEFT JOIN orders o ON o.user_id = u.id
GROUP BY u.id, u.name;

COUNT(o.id) jo COUNT(*) — perndryshe 0-order user del 1.

S7. Users me te pakten 2 porosi

SELECT user_id, COUNT(*) AS n
FROM orders
GROUP BY user_id
HAVING COUNT(*) >= 2;

S8. Mesatarja e totalit te porosive

SELECT AVG(total) FROM orders;

S9. Produkte qe nuk jane shitur kurre

SELECT p.name
FROM products p
LEFT JOIN order_items i ON i.product_id = p.id
WHERE i.id IS NULL;

S10. Top 5 produkte sipas qty te shitur

SELECT p.name, SUM(i.qty) AS sold
FROM products p
JOIN order_items i ON i.product_id = p.id
GROUP BY p.name
ORDER BY sold DESC
LIMIT 5;

S11. Email duplikat

SELECT email, COUNT(*) 
FROM users
GROUP BY email
HAVING COUNT(*) > 1;

S12. Xhiro per dite

SELECT created_at::date AS d, SUM(total)
FROM orders
GROUP BY 1
ORDER BY 1;
-- MySQL: DATE(created_at)

S13. Porosi te 7 diteve te fundit, status paid

SELECT * FROM orders
WHERE status = 'paid'
  AND created_at >= NOW() - INTERVAL '7 days';
-- MySQL: created_at >= NOW() - INTERVAL 7 DAY

S14. Ul stock me 1 nese ka stock

UPDATE products
SET stock = stock - 1
WHERE id = 9 AND stock > 0;

Kontrollo rowcount. Fol transaksion me INSERT order_items.

S15. Useri me spent me te larte

SELECT u.email, SUM(o.total) AS spent
FROM users u
JOIN orders o ON o.user_id = u.id
GROUP BY u.email
ORDER BY spent DESC
LIMIT 1;

S16. CASE: etiketo porosine

small < 20, medium < 100, else large.

SELECT id, total,
  CASE
    WHEN total < 20 THEN 'small'
    WHEN total < 100 THEN 'medium'
    ELSE 'large'
  END AS size
FROM orders;
